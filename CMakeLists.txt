cmake_minimum_required(VERSION 3.20)
project(Undertow VERSION 1.0 LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2)
find_package(OpenGL REQUIRED)
find_package(Freetype REQUIRED)
add_executable(undertow src/main.cpp src/motion.cpp src/renderer.cpp src/overlay.cpp)
target_compile_features(undertow PRIVATE cxx_std_20)
target_compile_options(undertow PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(undertow PRIVATE GL_GLEXT_PROTOTYPES)
target_link_libraries(undertow PRIVATE PkgConfig::SDL OpenGL::GL Freetype::Freetype)
include(CheckIPOSupported)
check_ipo_supported(RESULT IPO_OK)
if(IPO_OK)
  set_property(TARGET undertow PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
endif()
install(TARGETS undertow RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(motion_check tests/motion_check.cpp src/motion.cpp)
  target_include_directories(motion_check PRIVATE src)
  target_compile_features(motion_check PRIVATE cxx_std_20)
  add_test(NAME motion_stability COMMAND motion_check)
endif()
